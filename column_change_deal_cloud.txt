SELECT 
  COLUMN_NAME,
  DATA_TYPE,
  CHARACTER_MAXIMUM_LENGTH,
  NUMERIC_PRECISION,
  NUMERIC_SCALE,
  COMMENT
FROM dev_staging.information_schema.columns 
WHERE TABLE_SCHEMA = 'PUBLIC'
  AND TABLE_NAME = 'DEALCLOUD_DAILY_DEALS'
  AND COLUMN_NAME IN (
                    'COUNT_AS_NEW_DEAL',
                    'SOURCE_COMPANY',
                    'COMPANY_ATTACHMENT_LEVERAGE',
                    'MIN_WAL',
                    'MIN_IRR',
                    'COMPANY_EBITDA',
                    'ETL',
                    'COLLATERAL_TYPE',
                    'COMPANY_TOTAL_LEVERAGE',
                    'COMPANY_TOTAL_LTV',
                    'CAP_CHARGE',
                    'MIN_MOIC',
                    'COMPANY_EV_MULTIPLE',
                    'WEIGHTED_AVERAGE_LIFE',
                    'COMPANY_ATTACHMENT_LTV'
  )
ORDER BY COLUMN_NAME;

ALTER TABLE DEV_STAGING.PUBLIC.DEALCLOUD_DAILY_DEALS 
MODIFY COLUMN COUNT_AS_NEW_DEAL VARCHAR(16777216) 
COMMENT 'Flag indicating whether a transaction should be treated as a new origination for pipeline, KPI, and reporting purposes, as opposed to an amendment, add-on, refinancing, or restructuring.';
