WITH base_compare AS (
    SELECT
        b.full_date,
        b.aladdin_id,
        b.portfolio_code,
        b.pm_mandate,
        b.pm_ibu,
        b.pm_lmg,

        /* ================= MARKET VALUE ================= */
        COALESCE(a.MARKET_VALUE_BASE,0)  AS MARKET_VALUE_BASE_UAT,
        COALESCE(b.MARKET_VALUE_BASE,0)  AS MARKET_VALUE_BASE_PROD,
        ROUND(COALESCE(a.MARKET_VALUE_BASE,0) - COALESCE(b.MARKET_VALUE_BASE,0),10) AS MARKET_VALUE_BASE_DIFF,

        COALESCE(a.MARKET_VALUE_LOCAL,0) AS MARKET_VALUE_LOCAL_UAT,
        COALESCE(b.MARKET_VALUE_LOCAL,0) AS MARKET_VALUE_LOCAL_PROD,
        ROUND(COALESCE(a.MARKET_VALUE_LOCAL,0) - COALESCE(b.MARKET_VALUE_LOCAL,0),10) AS MARKET_VALUE_LOCAL_DIFF,

        COALESCE(a.MARKET_VALUE_USD,0)   AS MARKET_VALUE_USD_UAT,
        COALESCE(b.MARKET_VALUE_USD,0)   AS MARKET_VALUE_USD_PROD,
        ROUND(COALESCE(a.MARKET_VALUE_USD,0) - COALESCE(b.MARKET_VALUE_USD,0),10) AS MARKET_VALUE_USD_DIFF,

        /* ================= CLEAN MARKET VALUE ================= */
        COALESCE(a.CLEAN_MARKET_VALUE_BASE,0) AS CLEAN_MARKET_VALUE_BASE_UAT,
        COALESCE(b.CLEAN_MARKET_VALUE_BASE,0) AS CLEAN_MARKET_VALUE_BASE_PROD,
        ROUND(COALESCE(a.CLEAN_MARKET_VALUE_BASE,0) - COALESCE(b.CLEAN_MARKET_VALUE_BASE,0),10) AS CLEAN_MARKET_VALUE_BASE_DIFF,

        COALESCE(a.CLEAN_MARKET_VALUE_LOCAL,0) AS CLEAN_MARKET_VALUE_LOCAL_UAT,
        COALESCE(b.CLEAN_MARKET_VALUE_LOCAL,0) AS CLEAN_MARKET_VALUE_LOCAL_PROD,
        ROUND(COALESCE(a.CLEAN_MARKET_VALUE_LOCAL,0) - COALESCE(b.CLEAN_MARKET_VALUE_LOCAL,0),10) AS CLEAN_MARKET_VALUE_LOCAL_DIFF,

        COALESCE(a.CLEAN_MARKET_VALUE_USD,0) AS CLEAN_MARKET_VALUE_USD_UAT,
        COALESCE(b.CLEAN_MARKET_VALUE_USD,0) AS CLEAN_MARKET_VALUE_USD_PROD,
        ROUND(COALESCE(a.CLEAN_MARKET_VALUE_USD,0) - COALESCE(b.CLEAN_MARKET_VALUE_USD,0),10) AS CLEAN_MARKET_VALUE_USD_DIFF,

        /* ================= EXPOSURE ================= */
        COALESCE(a.EXPOSURE_BASE,0) AS EXPOSURE_BASE_UAT,
        COALESCE(b.EXPOSURE_BASE,0) AS EXPOSURE_BASE_PROD,
        ROUND(COALESCE(a.EXPOSURE_BASE,0) - COALESCE(b.EXPOSURE_BASE,0),10) AS EXPOSURE_BASE_DIFF,

        COALESCE(a.EXPOSURE_LOCAL,0) AS EXPOSURE_LOCAL_UAT,
        COALESCE(b.EXPOSURE_LOCAL,0) AS EXPOSURE_LOCAL_PROD,
        ROUND(COALESCE(a.EXPOSURE_LOCAL,0) - COALESCE(b.EXPOSURE_LOCAL,0),10) AS EXPOSURE_LOCAL_DIFF,

        COALESCE(a.EXPOSURE_USD,0) AS EXPOSURE_USD_UAT,
        COALESCE(b.EXPOSURE_USD,0) AS EXPOSURE_USD_PROD,
        ROUND(COALESCE(a.EXPOSURE_USD,0) - COALESCE(b.EXPOSURE_USD,0),10) AS EXPOSURE_USD_DIFF,

        /* ================= CURRENT FACE ================= */
        COALESCE(a.CURRENT_FACE_BASE,0) AS CURRENT_FACE_BASE_UAT,
        COALESCE(b.CURRENT_FACE_BASE,0) AS CURRENT_FACE_BASE_PROD,
        ROUND(COALESCE(a.CURRENT_FACE_BASE,0) - COALESCE(b.CURRENT_FACE_BASE,0),10) AS CURRENT_FACE_BASE_DIFF,

        COALESCE(a.CURRENT_FACE_LOCAL,0) AS CURRENT_FACE_LOCAL_UAT,
        COALESCE(b.CURRENT_FACE_LOCAL,0) AS CURRENT_FACE_LOCAL_PROD,
        ROUND(COALESCE(a.CURRENT_FACE_LOCAL,0) - COALESCE(b.CURRENT_FACE_LOCAL,0),10) AS CURRENT_FACE_LOCAL_DIFF,

        COALESCE(a.CURRENT_FACE_USD,0) AS CURRENT_FACE_USD_UAT,
        COALESCE(b.CURRENT_FACE_USD,0) AS CURRENT_FACE_USD_PROD,
        ROUND(COALESCE(a.CURRENT_FACE_USD,0) - COALESCE(b.CURRENT_FACE_USD,0),10) AS CURRENT_FACE_USD_DIFF,

        /* ================= UNFUNDED ================= */
        COALESCE(a.UNFUNDED_BASE,0) AS UNFUNDED_BASE_UAT,
        COALESCE(b.UNFUNDED_BASE,0) AS UNFUNDED_BASE_PROD,
        ROUND(COALESCE(a.UNFUNDED_BASE,0) - COALESCE(b.UNFUNDED_BASE,0),10) AS UNFUNDED_BASE_DIFF,

        COALESCE(a.UNFUNDED_USD,0) AS UNFUNDED_USD_UAT,
        COALESCE(b.UNFUNDED_USD,0) AS UNFUNDED_USD_PROD,
        ROUND(COALESCE(a.UNFUNDED_USD,0) - COALESCE(b.UNFUNDED_USD,0),10) AS UNFUNDED_USD_DIFF

    FROM SANDBOX_01.TEST_CAROLINE.CDST_IBOR_HOLDING_12_19_2025_BEFORE b
    LEFT JOIN SANDBOX_01.TEST_CAROLINE.CDST_IBOR_HOLDING_12_19_2025_AFTER a
        ON a.full_date = b.full_date
       AND a.asset_id = b.asset_id
       AND a.portfolio_id = b.portfolio_id
    WHERE b.full_date BETWEEN '2025-08-01' AND '2025-12-19'
)

SELECT *
FROM base_compare
WHERE
      ABS(MARKET_VALUE_BASE_DIFF) > 1
   OR ABS(MARKET_VALUE_LOCAL_DIFF) > 1
   OR ABS(MARKET_VALUE_USD_DIFF) > 1
   OR ABS(CLEAN_MARKET_VALUE_BASE_DIFF) > 1
   OR ABS(CLEAN_MARKET_VALUE_LOCAL_DIFF) > 1
   OR ABS(CLEAN_MARKET_VALUE_USD_DIFF) > 1
   OR ABS(EXPOSURE_BASE_DIFF) > 1
   OR ABS(EXPOSURE_LOCAL_DIFF) > 1
   OR ABS(EXPOSURE_USD_DIFF) > 1
   OR ABS(CURRENT_FACE_BASE_DIFF) > 1
   OR ABS(CURRENT_FACE_LOCAL_DIFF) > 1
   OR ABS(CURRENT_FACE_USD_DIFF) > 1
   OR ABS(UNFUNDED_BASE_DIFF) > 1
   OR ABS(UNFUNDED_USD_DIFF) > 1;
