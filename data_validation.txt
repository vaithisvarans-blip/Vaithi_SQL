WITH base_data AS (
    SELECT
        b.full_date,
        b.aladdin_id,
        b.portfolio_code,
        b.pm_mandate,
        b.pm_ibu,
        b.pm_lmg,
        b.asset_id,
        b.portfolio_id,

        /* ---------- PROD VALUES ---------- */
        COALESCE(b.MARKET_VALUE_BASE, 0)    AS MARKET_VALUE_BASE_PROD,
        COALESCE(b.MARKET_VALUE_LOCAL, 0)   AS MARKET_VALUE_LOCAL_PROD,
        COALESCE(b.MARKET_VALUE_USD, 0)     AS MARKET_VALUE_USD_PROD,
        COALESCE(b.EXPOSURE_BASE, 0)        AS EXPOSURE_BASE_PROD,
        COALESCE(b.EXPOSURE_USD, 0)         AS EXPOSURE_USD_PROD

    FROM prod_portfolio_intel.public.cdst_ibor_holding b
    WHERE b.full_date BETWEEN '2025-08-01' AND '2025-12-19'
),

uat_data AS (
    SELECT
        a.full_date,
        a.asset_id,
        a.portfolio_id,

        /* ---------- UAT VALUES ---------- */
        COALESCE(a.CLEAN_MARKET_VALUE_BASE, 0)                     AS CLEAN_MARKET_VALUE_BASE_UAT,
        COALESCE(a.SETTLEMENT_DATE_MARKET_VALUE_BASE, 0)           AS SETTLEMENT_DATE_MARKET_VALUE_BASE_UAT,
        COALESCE(a.SETTLEMENT_DATE_CLEAN_MARKET_VALUE_BASE, 0)     AS SETTLEMENT_DATE_CLEAN_MARKET_VALUE_BASE_UAT,
        COALESCE(a.END_OF_DAY_MARKET_VALUE_BASE, 0)                AS END_OF_DAY_MARKET_VALUE_BASE_UAT,

        COALESCE(a.CLEAN_MARKET_VALUE_LOCAL, 0)                    AS CLEAN_MARKET_VALUE_LOCAL_UAT,
        COALESCE(a.SETTLEMENT_DATE_CLEAN_MARKET_VALUE_LOCAL, 0)    AS SETTLEMENT_DATE_CLEAN_MARKET_VALUE_LOCAL_UAT,

        COALESCE(a.CLEAN_MARKET_VALUE_USD, 0)                      AS CLEAN_MARKET_VALUE_USD_UAT,
        COALESCE(a.SETTLEMENT_DATE_MARKET_VALUE_USD, 0)            AS SETTLEMENT_DATE_MARKET_VALUE_USD_UAT,
        COALESCE(a.SETTLEMENT_DATE_CLEAN_MARKET_VALUE_USD, 0)      AS SETTLEMENT_DATE_CLEAN_MARKET_VALUE_USD_UAT,
        COALESCE(a.END_OF_DAY_MARKET_VALUE_USD, 0)                 AS END_OF_DAY_MARKET_VALUE_USD_UAT,

        COALESCE(a.SETTLEMENT_DATE_EXPOSURE_BASE, 0)               AS SETTLEMENT_DATE_EXPOSURE_BASE_UAT,
        COALESCE(a.END_OF_DAY_EXPOSURE_BASE, 0)                    AS END_OF_DAY_EXPOSURE_BASE_UAT,

        COALESCE(a.SETTLEMENT_DATE_EXPOSURE_USD, 0)                AS SETTLEMENT_DATE_EXPOSURE_USD_UAT,
        COALESCE(a.END_OF_DAY_EXPOSURE_USD, 0)                     AS END_OF_DAY_EXPOSURE_USD_UAT

    FROM sandbox_01.private_credit.stage_cdst_ibor_holding a
),

wso_data AS (
    SELECT
        business_date,
        SHA2(cusip) AS asset_id,
        SHA2(DIM_PORTF_PORTFOLIO_CODE) AS portfolio_id,
        COALESCE(MARKETVALUETRADED, MARKETVALUESETTLED) AS WSO_MARKET_VALUE
    FROM SANDBOX_01.PRIVATE_CREDIT.VW_WSO_LOAN_POSITIONS_SUMMARY
)

SELECT
    b.full_date,
    b.aladdin_id,
    b.portfolio_code,
    b.pm_mandate,
    b.pm_ibu,
    b.pm_lmg,

    w.WSO_MARKET_VALUE,

    /* ---------- PROD ---------- */
    b.MARKET_VALUE_BASE_PROD,
    b.MARKET_VALUE_LOCAL_PROD,
    b.MARKET_VALUE_USD_PROD,
    b.EXPOSURE_BASE_PROD,
    b.EXPOSURE_USD_PROD,

    /* ---------- UAT ---------- */
    u.CLEAN_MARKET_VALUE_BASE_UAT,
    u.SETTLEMENT_DATE_MARKET_VALUE_BASE_UAT,
    u.SETTLEMENT_DATE_CLEAN_MARKET_VALUE_BASE_UAT,
    u.END_OF_DAY_MARKET_VALUE_BASE_UAT,

    u.CLEAN_MARKET_VALUE_LOCAL_UAT,
    u.SETTLEMENT_DATE_CLEAN_MARKET_VALUE_LOCAL_UAT,

    u.CLEAN_MARKET_VALUE_USD_UAT,
    u.SETTLEMENT_DATE_MARKET_VALUE_USD_UAT,
    u.SETTLEMENT_DATE_CLEAN_MARKET_VALUE_USD_UAT,
    u.END_OF_DAY_MARKET_VALUE_USD_UAT,

    u.SETTLEMENT_DATE_EXPOSURE_BASE_UAT,
    u.END_OF_DAY_EXPOSURE_BASE_UAT,
    u.SETTLEMENT_DATE_EXPOSURE_USD_UAT,
    u.END_OF_DAY_EXPOSURE_USD_UAT

FROM base_data b
LEFT JOIN uat_data u
       ON b.full_date     = u.full_date
      AND b.asset_id      = u.asset_id
      AND b.portfolio_id  = u.portfolio_id
LEFT JOIN wso_data w
       ON b.full_date     = w.business_date
      AND b.asset_id      = w.asset_id
      AND b.portfolio_id  = w.portfolio_id

WHERE w.WSO_MARKET_VALUE IS NOT NULL

/* ---------- VALIDATION LOGIC ---------- */
AND NOT (
       ROUND(b.MARKET_VALUE_LOCAL_PROD, 2) = ROUND(u.CLEAN_MARKET_VALUE_LOCAL_UAT, 2)
    OR ROUND(b.MARKET_VALUE_LOCAL_PROD, 2) = ROUND(u.SETTLEMENT_DATE_CLEAN_MARKET_VALUE_LOCAL_UAT, 2)
);
